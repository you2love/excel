import pandas as pd from openpyxl import Workbook # 方法 1: pandas 快速导出 df = pd.DataFrame({ "A": [1, 2, 3], "B": ["x", "y", "z"] }) df.to_excel("output.xlsx", index=False) # 方法 2: openpyxl 精细控制 wb = Workbook() ws = wb.active ws["A1"] = "姓名" ws.append(["张三", 28, "北京"]) wb.save("output.xlsx")
package main import "github.com/xuri/excelize/v2" func writeExcel() { f := excelize.NewFile() defer f.Close() // 设置单元格值 f.SetCellValue("Sheet1", "A1", "姓名") f.SetCellValue("Sheet1", "B1", "年龄") // 批量添加数据 data := [][]interface{}{ {"张三", 28}, {"李四", 32}, } for i, row := range data { for j, val := range row { cell, _ := excelize.CoordinatesToCellName(j+1, i+2) f.SetCellValue("Sheet1", cell, val) } } // 保存 f.SaveAs("output.xlsx") }
const ExcelJS = require('exceljs'); async function writeExcel() { const workbook = new ExcelJS.Workbook(); const worksheet = workbook.addWorksheet('Sheet1'); // 设置表头 worksheet.getRow(1).values = ['姓名', '年龄', '城市']; // 添加数据行 worksheet.addRows([ ['张三', 28, '北京'], ['李四', 32, '上海'] ]); // 设置列宽 worksheet.columns = [ { key: 'name', width: 15 }, { key: 'age', width: 10 }, { key: 'city', width: 15 } ]; await workbook.xlsx.writeFile('output.xlsx'); }
