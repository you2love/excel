🔄 批量处理

import pandas as pd
from pathlib import Path

def merge_excel_files(folder_path, output_file):
    """合并多个 Excel 文件"""
    all_data = []
    
    for file in Path(folder_path).glob("*.xlsx"):
        df = pd.read_excel(file)
        df["来源文件"] = file.name
        all_data.append(df)
    
    merged = pd.concat(all_data, ignore_index=True)
    merged.to_excel(output_file, index=False)
    return merged

def split_excel_by_column(input_file, column, output_folder):
    """按列值拆分 Excel 文件"""
    df = pd.read_excel(input_file)
    
    for value in df[column].unique():
        subset = df[df[column] == value]
        output = f"{output_folder}/{column}_{value}.xlsx"
        subset.to_excel(output, index=False)

# 使用示例
merge_excel_files("./monthly_reports", "year_summary.xlsx")
split_excel_by_column("sales.xlsx", "地区", "./regions")

import (
    "fmt"
    "log"
    
    "github.com/xuri/excelize/v2"
)

func main() {
    // 打开 Excel 文件
    f, err := excelize.OpenFile("data.xlsx")
    if err != nil {
        log.Fatal(err)
    }
    defer f.Close()

    // 获取默认工作表名
    sheetName := f.GetSheetName(0)
    fmt.Println("工作表:", sheetName)

    // 读取单元格
    cellValue, _ := f.GetCellValue(sheetName, "A1")
    fmt.Println("A1 的值:", cellValue)

    // 获取所有行
    rows, _ := f.GetRows(sheetName)
    for i, row := range rows {
        if i == 0 {
            fmt.Println("表头:", row)
            continue
        }
        fmt.Printf("第%d行: %v\n", i+1, row)
    }

    // 遍历列
    cols, _ := f.GetCols(sheetName)
    for i, col := range cols {
        fmt.Printf("第%d列: %v\n", i+1, col)
    }
}

import (
    "fmt"
    "sync"
    "github.com/xuri/excelize/v2"
)

func processLargeExcel(filename string) {
    f, _ := excelize.OpenFile(filename)
    defer f.Close()

    rows, _ := f.GetRows("Sheet1")
    var wg sync.WaitGroup
    resultChan := make(chan []string, 100)

    // 启动工作池
    for i := 0; i < 4; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            for row := range resultChan {
                // 处理数据
                fmt.Println("处理:", row)
            }
        }()
    }

    // 发送数据到通道
    for i, row := range rows {
        if i == 0 { continue } // 跳过表头
        resultChan <- row
    }

    close(resultChan)
    wg.Wait()
}

func mergeExcelFiles(files []string, output string) {
    result := excelize.NewFile()
    defer result.Close()

    var wg sync.WaitGroup
    dataChan := make(chan [][]string, len(files))

    // 并发读取文件
    for _, file := range files {
        wg.Add(1)
        go func(f string) {
            defer wg.Done()
            excel, _ := excelize.OpenFile(f)
            defer excel.Close()
            rows, _ := excel.GetRows("Sheet1")
            dataChan <- rows
        }(file)
    }

    go func() {
        wg.Wait()
        close(dataChan)
    }()

    // 合并数据
    rowNum := 1
    for rows := range dataChan {
        for _, row := range rows {
            for col, val := range row {
                cell, _ := excelize.CoordinatesToCellName(col+1, rowNum)
                result.SetCellValue("Sheet1", cell, val)
            }
            rowNum++
        }
    }

    result.Save(output)
}

🌐 Web 服务集成

package main

import (
    "bytes"
    "net/http"
    "github.com/gin-gonic/gin"
    "github.com/xuri/excelize/v2"
)

func main() {
    r := gin.Default()

    // 上传 Excel 并解析
    r.POST("/upload", func(c *gin.Context) {
        file, _ := c.FormFile("excel")
        f, _ := excelize.OpenReader(file.Filename)
        defer f.Close()

        rows, _ := f.GetRows("Sheet1")
        c.JSON(http.StatusOK, gin.H{
            "data": rows,
        })
    })

    // 生成并下载 Excel
    r.GET("/export", func(c *gin.Context) {
        f := excelize.NewFile()
        defer f.Close()

        f.SetCellValue("Sheet1", "A1", "姓名")
        f.SetCellValue("Sheet1", "B1", "年龄")
        f.SetCellValue("Sheet1", "A2", "张三")
        f.SetCellValue("Sheet1", "B2", 28)

        buf := new(bytes.Buffer)
        f.Write(buf)

        c.Header("Content-Disposition", 
            "attachment; filename=export.xlsx")
        c.Data(http.StatusOK, 
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
            buf.Bytes())
    })

    r.Run(":8080")
}

3. Python vs Go 对比总结

维度 Python Go
学习曲线 ⭐⭐ 简单易学，语法友好 ⭐⭐⭐ 需要理解类型系统
执行速度 ⭐⭐ 解释型，相对较慢 ⭐⭐⭐⭐⭐ 编译型，性能优异
数据处理 ⭐⭐⭐⭐⭐ pandas 生态强大 ⭐⭐⭐ 需要手动处理
并发能力 ⭐⭐ GIL 限制 ⭐⭐⭐⭐⭐ goroutine 轻量并发
部署便利 ⭐⭐ 需要 Python 环境 ⭐⭐⭐⭐⭐ 单二进制文件
适用场景 数据分析、机器学习、快速原型 高性能服务、大规模处理、微服务
选择建议：
• 数据分析、机器学习、快速开发 → 选择 Python
• 高性能 Web 服务、大规模并发处理 → 选择 Go
• 两者结合：Python 做数据分析，Go 做 API 服务
